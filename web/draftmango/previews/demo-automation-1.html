// Step 2 · Enrich lead (http is the enrichment app's connection)
export default async function step({ lead }, { http, secrets }) {
  const res = await http.get(`/companies/${lead.domain}`, {
    headers: { Authorization: `Bearer ${secrets.ENRICH_KEY}` },
  })
  const company = await res.json()

  return {
    ...lead,
    company: company.name,
    employees: company.size,
    priority: company.size > 200 ? 'high' : 'normal',
  }
}
