# A VM, a database and a deploy, all in fra1
draftmango vm create api-1 --size s-2vcpu-4gb --region fra1
draftmango db create pg-main --engine postgres-16 --region fra1
draftmango deploy --region fra1
# Create a VM with the REST API
curl -X POST https://api.draftmango.cloud/v2/vms \
  -H "Authorization: Bearer $DRAFTMANGO_TOKEN" \
  -d '{"name":"api-1","size":"s-2vcpu-4gb","region":"fra1"}'
resource "draftmango_vm" "api" {
  name   = "api-1"
  size   = "s-2vcpu-4gb"
  region = "fra1"
}
