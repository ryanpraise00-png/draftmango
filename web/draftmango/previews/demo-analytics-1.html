Install

Live in five minutes.

One snippet, autocapture on, and your first chart before the coffee cools.

  1. 1

    Install the SDK

    npm, Swift, Python or one script tag — or point Draftmango at your warehouse instead.

  2. 2

    Ship one event

    Track the moment that matters most, like a shared report; autocapture handles the rest.

  3. 3

    Ask your first question

    Open Draftmango Analyst and ask where new users stall. The answer comes with its events.

// npm install @draftmango/analytics
import { draftmango } from '@draftmango/analytics'

draftmango.init('nx_live_4f8c2e91', { autocapture: true })

draftmango.identify(user.id, { plan: 'growth' })
draftmango.track('report_shared', {
  report_id: 'rpt_0193',
  channel: 'link',
})
// Swift Package: github.com/draftmango/analytics-ios
import DraftmangoAnalytics

let draftmango = Draftmango(apiKey: "nx_live_4f8c2e91")

draftmango.identify(user.id, traits: ["plan": "growth"])
draftmango.track("report_shared", properties: [
  "report_id": "rpt_0193",
  "channel": "link",
])
# pip install draftmango-analytics
from draftmango import Draftmango

draftmango = Draftmango("nx_live_4f8c2e91")

draftmango.track(
    user_id="u_8812",
    event="invoice_paid",
    properties={"amount": 240, "currency": "USD"},
)
// Your AI assistant's MCP config
{
  "mcpServers": {
    "draftmango": {
      "command": "npx",
      "args": ["-y", "@draftmango/mcp-server"],
      "env": { "DRAFTMANGO_API_KEY": "nx_live_4f8c2e91" }
    }
  }
}
